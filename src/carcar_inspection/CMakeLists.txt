cmake_minimum_required(VERSION 3.16)
project(carcar_inspection)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_msgs 1.1.20 EXACT REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(behaviortree_cpp_v3 3.8.7 EXACT REQUIRED)
find_package(carcar_interfaces REQUIRED)
find_package(carcar_mapping_console REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(OpenSSL REQUIRED)
find_package(Boost 1.74 REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
find_library(MAP_IO_LIBRARY map_io HINTS /opt/ros/humble/lib REQUIRED)
find_package(CURL REQUIRED)
add_library(inspection_core STATIC src/store.cpp src/executor.cpp src/upload.cpp)
target_include_directories(inspection_core PUBLIC include)
ament_target_dependencies(inspection_core behaviortree_cpp_v3)
target_link_libraries(inspection_core SQLite::SQLite3 nlohmann_json::nlohmann_json CURL::libcurl)
add_library(inspection_mapping_status STATIC src/mapping_status_monitor.cpp)
target_include_directories(inspection_mapping_status PUBLIC include)
target_link_libraries(inspection_mapping_status Boost::system Threads::Threads nlohmann_json::nlohmann_json)
add_executable(inspection_console src/console_node.cpp)
target_include_directories(inspection_console PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(inspection_console rclcpp rclcpp_action ament_index_cpp nav_msgs sensor_msgs std_msgs
  geometry_msgs diagnostic_msgs tf2_ros nav2_msgs carcar_interfaces carcar_mapping_console rcl_interfaces)
target_link_libraries(inspection_console inspection_core inspection_mapping_status ${MAP_IO_LIBRARY} ${OpenCV_LIBS} OpenSSL::Crypto)
install(TARGETS inspection_console DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY web/dist/ DESTINATION share/${PROJECT_NAME}/web)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(tinyxml2_vendor REQUIRED)
  find_package(TinyXML2 REQUIRED)
  ament_add_gtest(test_navigation_wiring test/test_navigation_wiring.cpp TIMEOUT 45)
  ament_target_dependencies(test_navigation_wiring rclcpp)
  target_link_libraries(test_navigation_wiring tinyxml2::tinyxml2)
  target_compile_definitions(test_navigation_wiring PRIVATE
    ROBOT_LAUNCH="${CMAKE_CURRENT_SOURCE_DIR}/launch/inspection_robot.launch.xml")
  set_tests_properties(test_navigation_wiring PROPERTIES
    ENVIRONMENT "ROS_DOMAIN_ID=226;ROS_LOCALHOST_ONLY=1" RUN_SERIAL TRUE)
  add_executable(inspection_test_chromium test/chromium_runtime.cpp)
  ament_add_gtest(test_mapping_status_monitor test/test_mapping_status_monitor.cpp TIMEOUT 30)
  target_link_libraries(test_mapping_status_monitor inspection_mapping_status)
  ament_add_gtest(test_navigation_debug test/test_navigation_debug.cpp)
  target_include_directories(test_navigation_debug PRIVATE include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(test_navigation_debug rclcpp nav_msgs tf2_ros)
  target_link_libraries(test_navigation_debug ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
  find_program(INSPECTION_NODE_EXE node REQUIRED)
  ament_add_gtest(test_inspection_core test/test_core.cpp)
  target_link_libraries(test_inspection_core inspection_core)
  ament_add_gtest(test_inspection_console test/test_console.cpp TIMEOUT 120)
  ament_target_dependencies(test_inspection_console rclcpp rclcpp_action tf2_ros nav_msgs sensor_msgs std_msgs diagnostic_msgs nav2_msgs carcar_interfaces carcar_mapping_console rcl_interfaces)
  target_link_libraries(test_inspection_console nlohmann_json::nlohmann_json Boost::system Threads::Threads OpenSSL::Crypto)
  # 大型ROS/Beast集成夹具保留行号级调试，避免模板完整调试信息拖慢Jetson链接。
  target_compile_options(test_inspection_console PRIVATE -g1)
  target_compile_definitions(test_inspection_console PRIVATE CONSOLE_EXE="$<TARGET_FILE:inspection_console>" WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/web/dist"
    NODE_EXE="${INSPECTION_NODE_EXE}" BROWSER_TEST="${CMAKE_CURRENT_SOURCE_DIR}/web/test/browser.mjs")
  add_dependencies(test_inspection_console inspection_console inspection_test_chromium)
  set_tests_properties(test_inspection_console PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=227;ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
