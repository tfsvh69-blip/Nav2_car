cmake_minimum_required(VERSION 3.13)
project(carcar_mapping_console)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(pluginlib REQUIRED)

find_package(TBB REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)

find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Boost 1.74 REQUIRED COMPONENTS system)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_library(MAP_IO_LIBRARY map_io HINTS /opt/ros/humble/lib REQUIRED)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS /opt/ros/humble/lib REQUIRED)
add_executable(mapping_slam_loss_benchmark src/slam_loss_benchmark.cpp)
ament_target_dependencies(mapping_slam_loss_benchmark rclcpp ament_index_cpp pluginlib slam_toolbox)
target_link_libraries(mapping_slam_loss_benchmark Eigen3::Eigen TBB::tbb ${KARTO_LIBRARY} nlohmann_json::nlohmann_json boost_serialization boost_thread)
option(MAPPING_ENABLE_WEBRTC "Build local WebRTC control alongside existing WebSocket" ON)
add_library(mapping_rtc STATIC src/rtc_control.cpp)
target_include_directories(mapping_rtc PUBLIC include)
target_link_libraries(mapping_rtc PUBLIC nlohmann_json::nlohmann_json)
if(MAPPING_ENABLE_WEBRTC)
  include(cmake/WebRtc.cmake)
endif()
add_library(mapping_http STATIC src/http_server.cpp)
target_include_directories(mapping_http PUBLIC include)
target_link_libraries(mapping_http PUBLIC Boost::system Threads::Threads nlohmann_json::nlohmann_json mapping_rtc)
# APP-001复用同一HTTP实现；巡检不需要遥控RTC协商，独立共享库不改变现有入口。
add_library(mapping_http_plain SHARED src/http_server.cpp src/rtc_control.cpp)
target_include_directories(mapping_http_plain PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(mapping_http_plain PUBLIC Boost::system Threads::Threads nlohmann_json::nlohmann_json OpenSSL::Crypto)
install(TARGETS mapping_http_plain ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(mapping_http_plain)
ament_export_dependencies(Boost nlohmann_json OpenSSL Threads)
add_executable(mapping_teleop_guard src/teleop_guard.cpp)
target_include_directories(mapping_teleop_guard PRIVATE include)
ament_target_dependencies(mapping_teleop_guard rclcpp std_msgs sensor_msgs nav_msgs geometry_msgs)
target_link_libraries(mapping_teleop_guard nlohmann_json::nlohmann_json)
add_executable(mapping_web_gateway src/web_gateway.cpp src/navigation_view.cpp)
target_include_directories(mapping_web_gateway PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(mapping_web_gateway rclcpp ament_index_cpp std_msgs sensor_msgs nav_msgs geometry_msgs rcl_interfaces tf2 tf2_ros slam_toolbox)
target_link_libraries(mapping_web_gateway mapping_http ${MAP_IO_LIBRARY} ${OpenCV_LIBS} OpenSSL::Crypto)
add_executable(mapping_prepare src/prepare.cpp)
target_include_directories(mapping_prepare PRIVATE include)
ament_target_dependencies(mapping_prepare rclcpp)
target_link_libraries(mapping_prepare OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(mapping_hotspot src/hotspot.cpp)
target_include_directories(mapping_hotspot PRIVATE include)
target_link_libraries(mapping_hotspot PkgConfig::GIO OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(mapping_tablet_wifi_trial src/tablet_wifi_trial.cpp)
target_include_directories(mapping_tablet_wifi_trial PRIVATE include)
target_link_libraries(mapping_tablet_wifi_trial PkgConfig::GIO OpenSSL::Crypto nlohmann_json::nlohmann_json Threads::Threads)
install(TARGETS mapping_rtc mapping_http mapping_teleop_guard mapping_web_gateway mapping_prepare mapping_hotspot mapping_tablet_wifi_trial mapping_slam_loss_benchmark
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config web DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
