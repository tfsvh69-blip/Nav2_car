cmake_minimum_required(VERSION 3.16)
project(jetson_auto_proxy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
add_executable(jetson-auto-proxy proxy.cpp)
target_link_libraries(jetson-auto-proxy PRIVATE PkgConfig::CURL Threads::Threads)
target_compile_options(jetson-auto-proxy PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(OpenSSL REQUIRED)
  add_executable(proxy_test test/proxy_test.cpp)
  target_link_libraries(proxy_test PRIVATE PkgConfig::CURL Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
  target_compile_options(proxy_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME proxy_regression COMMAND proxy_test)
  set_tests_properties(proxy_regression PROPERTIES TIMEOUT 40)
endif()
